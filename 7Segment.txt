#include <reg51.h>

sbit D1 = P3^0;
sbit D2 = P3^1;

unsigned char code seg[] = {
    0x3F, 0x06, 0x5B, 0x4F, 0x66,
	
    0x6D, 0x7D, 0x07, 0x7F, 0x6F
};

void delay()
{
    int i;
    for(i = 0; i < 1000; i++);
}

void main()
{
    int count, r;

    count = 0;

    while(1)
    {
        for(r = 0; r < 100; r++)
        {
            D1 = 0;
            D2 = 0;

            P1 = seg[count / 10];
            D1 = 1;
            delay();
            D1 = 0;

            P1 = seg[count % 10];
            D2 = 1;
            delay();
            D2 = 0;
        }

        count++;

        if(count > 99)
        {
            count = 0;
        }
    }
}
