#include <reg51.h>

#define DAC P1

void delay()
{
    unsigned int i, j;

    for(i = 0; i < 10; i++)
    {
        for(j = 0; j < 120; j++);
    }
}

void main()
{
    while(1)
    {
        
        DAC = 0xFF;
        delay();

      
        DAC = 0x00;
        delay();
    }
}
